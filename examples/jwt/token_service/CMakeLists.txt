# Create a CMake target for generating a *.idl.cpp.h header file
# for a defined IDL file using the nkppmeta compiler.
add_nk_idl (jwt_example_token_service_idl "${NK_RESOURCES}/jwt_example/IToken.idl"
            NK_MODULE "jwt_example"
            LANG      "CXX")

# Create a CMake target for generating a *.edl.cpp.h file
# for a defined EDL file using the nkppmeta compiler.
add_nk_edl (jwt_example_token_service "${NK_RESOURCES}/jwt_example/TokenService.edl"
            NK_MODULE "jwt_example"
            LANG      "CXX")

# Bind CMake targets generated by the add_nk_idl(), add_nk_cdl(), and add_nk_edl() commands
# to the main CMake target for building the application.
bind_nk_targets (jwt_example_token_service jwt_example_token_service_idl)

add_executable (TokenService
                src/main.cpp)

bind_nk_targets (TokenService jwt_example_token_service)
target_compile_definitions (TokenService PRIVATE AppTag="TokenService")

add_dependencies (TokenService openssl_headers)
target_link_libraries (TokenService PRIVATE
                       cpp-jwt::cpp-jwt
                       logrr_cpp::logrr_cpp-static
                       vfs::client)

# Disable the default VFS entity that is automatically provided by vfs::client.
set_target_properties (TokenService PROPERTIES vfs_entity::entity_REPLACEMENT "")
